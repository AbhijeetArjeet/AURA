object InfoFetcher {
    fun fetch(url: String): VideoInfo {
        val py = Python.getInstance()
        val ytdlp = py.getModule("yt_dlp")
        val builtins = py.getBuiltins()

        val options = builtins.callAttr("dict")
        options.callAttr("__setitem__", "noplaylist", true)
        options.callAttr("__setitem__", "quiet", true)
        options.callAttr("__setitem__", "simulate", true)

        val ytdlpObj = ytdlp.callAttr("YoutubeDL", options)
        val infoDict = ytdlpObj.callAttr("extract_info", url, false)

        if (infoDict == null || infoDict.toString() == "None") {
            throw Exception("yt-dlp could not extract information from this URL")
        }

        fun getStr(key: String): String {
            val v = infoDict.callAttr("get", key)
            return if (v == null || v.toString() == "None") "" else v.toString()
        }

        fun getLong(key: String): Long {
            val v = infoDict.callAttr("get", key)
            return if (v == null || v.toString() == "None") 0L else v.toLong()
        }

        return VideoInfo(
            url              = url,
            title            = getStr("title").ifBlank { "Unknown Video" },
            channel          = getStr("uploader").ifBlank { getStr("channel") },
            durationSeconds  = getLong("duration"),
            thumbnailUrl     = getStr("thumbnail"),
            viewCount        = getLong("view_count"),
        )
    }
}
